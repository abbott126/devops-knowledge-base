grep

从文件中根据需求过滤关键字所在的行

语法：

grep [OPTIONS] PATTERN [FILE...]
grep [OPTIONS] [-e PATTERN | -f FILE] [FILE...]

Example 1: 过滤root关键字所在的行

[root@mysql-master opt]# grep 'root'  passwd 
root:x:0:0:root:/root:/bin/bash
operator:x:11:0:operator:/root:/sbin/nologin

Example 2: 以^开头的行

[root@mysql-master opt]# grep   ^SELINUX  /etc/selinux/config 
SELINUX=enforcing
SELINUXTYPE=targeted

Example 3: 以$结尾的关键所在的行

[root@mysql-master opt]# grep   enforcing$  /etc/selinux/config 
SELINUX=enforcing

Example 4: 

[root@mysql-master opt]# grep  -E   '^SE|g$' /etc/selinux/config 
SELINUX=enforcing
SELINUXTYPE=targeted
[root@mysql-master opt]# grep  -E   '^SELINUX|g$' /etc/selinux/config 
SELINUX=enforcing
SELINUXTYPE=targeted 
[root@mysql-master opt]# grep  -E   '^SELINUX=|g$' /etc/selinux/config 
SELINUX=enforcing

Example 5: 忽略大小写 

[root@mysql-master opt]# grep  -Ei   '^selinux=|g$' /etc/selinux/config 
SELINUX=enforcing
[root@mysql-master opt]# egrep  -i   '^selinux=|g$' /etc/selinux/config 
SELINUX=enforcing

Awk

文本和数据进行处理的编程语言

[root@mysql-master opt]# ip ad | grep -E 'eth0$'     
    inet 192.168.122.134/24 brd 192.168.122.255 scope global noprefixroute dynamic eth0
[root@mysql-master opt]# ip ad | grep -E 'eth0$'      | awk  '{print  $1}'

[root@mysql-master opt]# ip ad | grep -E 'eth0$'      | awk  '{print  $2}'

[root@mysql-master opt]# ip ad | grep -E 'eth0$'      | awk  '{print  $2}'  | awk  -F/  '{print "总行:" NR}{print NF}{print $1}{print $2}{print  $0}'

#!/bin/awk   #告诉脚本使用什么语言解释
{
    for (i = NF; i <= NF; i--) {
        if ($i == "人") {  # 注意这里应该用==做比较，而不是=赋值
            print "hello world"
            break  # 找到后可以退出循环
        }
                else {
                    print $i
                }   
    }
}
