准备工作

修改主机名

[root@localhost ~]# hostnamectl  set-hostname web-server.com
[root@localhost ~]# bash
[root@web-server ~]#

获取软件包：

切记：一定官方网站获取

wget  https://dlcdn.apache.org//apr/apr-1.7.6.tar.gz
wget  https://dlcdn.apache.org//apr/apr-util-1.6.3.tar.gz
wget  https://dlcdn.apache.org/httpd/httpd-2.4.65.tar.gz

如果： 在工作的时候不能连接外网；

  1. 下载本地；使用SCP命令

创建管理目录

注意：

学习：源码安装软件包位置：/usr/local/

生产： 根据公司定义软件包安装位置 /app

[root@web-server ~]# mkdir  /usr/local/apache1
[root@web-server ~]# mkdir   /usr/local/apache1/{apr,apr-util,http}
[root@web-server ~]# ls /usr/local/apache1/
apr  apr-util  http

依次解压软件包

[root@web-server ~]# tar  xf  apr-1.7.6.tar.gz  -C  /usr/local/apache1/apr
[root@web-server ~]# tar  xf  apr-util-1.6.3.tar.gz  -C /usr/local/apache1/apr-util/
[root@web-server ~]# tar  xf  httpd-2.4.65.tar.gz  -C /usr/local/apache1/http/

依次安装软件包

对于源码部署：（口诀三板斧）

1. 检查环境
./configure
2. 编译
make
3. 编译安装
make install
  • apr
[root@web-server apr-1.7.6]# ./configure  --prefix=/usr/local/apache1/apr
[root@web-server apr-1.7.6]#  make 
[root@web-server apr-1.7.6]#  make install

判断上次执行的命令是否成功： echo $?

0 # 非0 表示执行失败

  • apr-util
[root@web-server apr-util-1.6.3]# ./configure  --prefix=/usr/local/apache1/apr-util   --with-apr=/usr/local/apache1/apr
[root@web-server apr-util-1.6.3]# make
[root@web-server apr-util-1.6.3]# make install

--prefix= # 指定安装目录

--with-apr= # 指定apr安装的位置

  • httpd
[root@web-server httpd-2.4.65]# ./configure \
--prefix=/usr/local/apache1/http/ \
--with-apr=/usr/local/apache1/apr \
--with-apr-util=/usr/local/apache1/apr-util/
[root@web-server httpd-2.4.65]# make 
[root@web-server httpd-2.4.65]# make install
  • 开启http服务
[root@web-server http]# bin/apachectl

基于IP地址创建虚拟主机

注意： 添加一块网卡

  1. 修改配置(打开支持引入虚拟主机配置文件)
# Virtual hosts
Include conf/extra/httpd-vhosts.conf

  1. 定义虚拟主机

  2. 创建项目

    [root@web-server ~]# mkdir /usr/local/apache1/http/htdocs/test-project 
[root@web-server ~]# vim   /usr/local/apache1/http/htdocs/test-project/index.html 
<html>
<head>
    <meta charset="UTF-8">
    <!-- 其他头部内容 -->
</head>

<h1>hello vhost 基于IP </h1>
</html>

  3. 配置虚拟主机

    [root@web-server ~]#vim  /usr/local/apache1/http/conf/extra/httpd-vhosts.conf 

<VirtualHost 192.168.19.120>
    #ServerAdmin webmaster@dummy-host2.example.com
    #定义项目目录
    DocumentRoot "htdocs/test-project"
    #ServerName dummy-host2.example.com
    定义日志
    ErrorLog "logs/test-project-error_log"
    CustomLog "logs/test-project-access_log" common
</VirtualHost>

  4. 重启服务

[root@web-server ~]# /usr/local/apache1/http/bin/apachectl  -k stop ^C
[root@web-server ~]# 
[root@web-server ~]# /usr/local/apache1/http/bin/apachectl
